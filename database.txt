  updated_at timestamp with time zone not null default now(),

  constraint trip_participant_locations_participant_id_fkey foreign KEY (participant_id) references trip_participants (id) on delete CASCADE,
  constraint trip_participant_locations_accuracy_m_check check (
    (
      (accuracy_m is null)
      or (
        (accuracy_m >= (0)::double precision)
        and (accuracy_m <= (100000)::double precision)
      )
    )
  ),
  constraint trip_participant_locations_latitude_check check (
    (
      (latitude >= ('-90'::integer)::double precision)
      and (latitude <= (90)::double precision)
    )
  ),
  constraint trip_participant_locations_longitude_check check (
    (
      (longitude >= ('-180'::integer)::double precision)
      and (longitude <= (180)::double precision)
    )
  )
) TABLESPACE pg_default;

select
  d.trip_date,
  d.label as day_label,

  s.id as stop_id,
  s.time_label,
  s.title,
